MaSV -> TenSV                            : TRUE

# Thuoc tinh trung gian: MaNganh, MaKhoa, MaHP, MaLHP, HocKy, Thu, TietBD, MaGV, MaPhong

# ==== 4. DANG CHUAN ====
Dang chuan cao nhat cua U: 1NF
Vi pham 2NF (phu thuoc rieng phan vao khoa, ve phai la thuoc tinh khong khoa): 4
   MaSV -> TenSV, NgaySinh, MaNganh
   MaLHP -> MaHP, HocKy, MaGV, MaPhong, Thu, TietBD
   MaGV -> TenGV, HocVi
   MaPhong -> ToaNha, SucChua
Vi pham 3NF: 7 (4 cai tren + MaNganh -> ..., MaKhoa -> ..., MaHP -> ...)
Vi pham BCNF: 9 (7 cai tren + 2 PTH lich: {MaPhong,HocKy,Thu,TietBD} -> MaLHP va
   {MaGV,HocKy,Thu,TietBD} -> MaLHP; ve trai cua chung khong phai sieu khoa vi bao dong khong chua MaSV, Diem)
# Chi con 2 PTH khong vi pham BCNF/3NF la: MaSV, MaLHP -> Diem (ve trai la khoa).
