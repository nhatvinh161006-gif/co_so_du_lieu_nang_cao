    3. TRANH CHẤP ĐỒNG THỜI 2 PHIÊN (sql/07_Transactions_va_Concurrency.sql)
    
        
    Phiên A vs Phiên B cạnh tranh suất cuối (SELECT ... FOR UPDATE)
    
        
    -- Chuẩn bị: đặt SiSoToiDa = 7 khi LHP2401001 đã có 6 đăng ký
UPDATE qldt.LopHocPhan SET SiSoToiDa = 7 WHERE MaLHP = 'LHP2401001';

-- [T1 - Phiên A] Giữ khóa dòng LHP2401001, đăng ký thành công (7/7), chưa COMMIT
BEGIN; CALL qldt.sp_DangKyHocPhan('SV2001004', 'LHP2401001', NULL, NULL);

-- [T2 - Phiên B] Bị chặn chờ khóa FOR UPDATE của Phiên A
BEGIN; CALL qldt.sp_DangKyHocPhan('SV2001003', 'LHP2401001', NULL, NULL);

-- [T3 - Phiên A] COMMIT nhả khóa -> [T4 - Phiên B] đếm thấy 7/7 -> từ chối an toàn.
    
      
    

      
    
        
    4. THỰC NGHIỆM ĐO HIỆU NĂNG (sql/08_Explain_Analyze_ThucNghiem.sql)
    
        
    So sánh EXPLAIN (ANALYZE, BUFFERS): Seq Scan vs Index Scan
    
        
    -- 1a. Tắt Index Scan để đo Sequential Scan
SET enable_indexscan = off; SET enable_bitmapscan = off;
EXPLAIN (ANALYZE, BUFFERS) SELECT MaSV, MaLHP, Diem FROM qldt.DangKy WHERE Diem >= 8.0;

-- 1b. Bật lại Index Scan (Partial Index idx_dangky_diem)
SET enable_indexscan = on; SET enable_bitmapscan = on;
EXPLAIN (ANALYZE, BUFFERS) SELECT MaSV, MaLHP, Diem FROM qldt.DangKy WHERE Diem >= 8.0;
